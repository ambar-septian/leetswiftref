```swift
import Collections

class Solution {
    struct Node: Comparable {
        let destination: Int
        let weight: Int

        init(_ destination: Int, _ weight: Int) {
            self.destination = destination
            self.weight = weight
        }

        static func < (_ lhs: Self, _ rhs: Self) -> Bool {
            lhs.weight < rhs.weight
        }
    }

    func networkDelayTime(_ times: [[Int]], _ n: Int, _ k: Int) -> Int {
        var adjList = [Int: [Node]]()

        for time in times {
            adjList[time[0], default: []].append(Node(time[1], time[2]))
        }

        var heap = Heap<Node>()
        heap.insert(Node(k, 0))

        var result = 0
        var visited = Set<Int>()

        while !heap.isEmpty {
            guard let node = heap.popMin(), !visited.contains(node.destination) else {
                continue
            }

            visited.insert(node.destination)
            result = max(result, node.weight)

            let neighbors = adjList[node.destination, default: []]
            for neighbor in neighbors {
                if visited.contains(neighbor.destination) {
                    continue
                }

                heap.insert(Node(neighbor.destination, node.weight + neighbor.weight))
            }
        }

        if visited.count != n {
            return -1
        }

        return result
    }
}
```
